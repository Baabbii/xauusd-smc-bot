"""
Script operațional -- rulează periodic (ex: la fiecare 5-15 minute prin
cron / Task Scheduler) pentru a detecta semnale NOI și a trimite alertă.

Acest script NU execută tranzacții -- doar te alertează. Decizia și
execuția rămân la tine (sau le automatizezi separat, cu multă grijă,
folosind API-ul brokerului tău).

De ce nu rulează pe GitHub Actions:
    GitHub Actions rulează pe servere Linux efemere, fără acces la
    terminalul MT5 (Windows-only) sau la contul tău de broker. Trebuie
    rulat local, pe mașina ta sau pe un VPS Windows unde ai MT5 deschis
    și poți exporta date live.

Flux tipic:
    1. Un script separat (al tău, specific brokerului) exportă cele mai
       recente date M1 în data/XAUUSD_M1.csv, suprascriind sau adăugând
       la fișierul existent.
    2. daily_run.py citește acel fișier, generează semnale, compară cu
       ultimul semnal deja alertat (salvat în state.json) și trimite
       alertă DOAR pentru semnalele noi.

Utilizare:
    python daily_run.py --m1 data/XAUUSD_M1.csv
"""

import argparse
import json
import os

from config import Config
from data_loader import load_csv, load_multi_timeframe
from strategy import generate_signals
from alerts import alert_new_signals

STATE_FILE = "state.json"


def load_state() -> dict:
    if os.path.exists(STATE_FILE):
        with open(STATE_FILE) as f:
            return json.load(f)
    return {"last_alerted_time": None}


def save_state(state: dict) -> None:
    with open(STATE_FILE, "w") as f:
        json.dump(state, f)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--m1", required=True)
    parser.add_argument("--m5", default=None)
    parser.add_argument("--m15", default=None)
    args = parser.parse_args()

    cfg = Config()
    state = load_state()

    if args.m5 and args.m15:
        m1 = load_csv(args.m1)
        m5 = load_csv(args.m5)
        m15 = load_csv(args.m15)
    else:
        tf = load_multi_timeframe(args.m1)
        m1, m5, m15 = tf["M1"], tf["M5"], tf["M15"]

    signals = generate_signals(m15, m5, m1, cfg)

    last_time = state.get("last_alerted_time")
    if last_time:
        new_signals = [s for s in signals if str(s.time) > last_time]
    else:
        # prima rulare: nu trimitem retroactiv toate semnalele istorice,
        # doar marcăm punctul de start
        new_signals = []

    if new_signals:
        print(f"Semnale noi găsite: {len(new_signals)}")
        alert_new_signals(new_signals)
    else:
        print("Niciun semnal nou.")

    if signals:
        state["last_alerted_time"] = str(signals[-1].time)
        save_state(state)


if __name__ == "__main__":
    main()
